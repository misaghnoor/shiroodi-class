    if(n>=th.excellent)return state.levels[0];if(n>=th.good)return state.levels[1];if(n>=th.acceptable)return state.levels[2];return state.levels[3];}
  return state.levels.find(l=>l.key===g)||null;
}
function formatGrade(aid,sub,sid){
  const g=getRawGrade(aid,sub,sid);if(g==null)return '—';
  if(isNumericMode())return fmtNum(g);
  const l=state.levels.find(x=>x.key===g);
  return l?l.icon+' '+l.label:'—';
}
function studentStats(sid,assessList){
  const list=assessList||state.assessments;
  let sum=0,cnt=0,filled=0,possible=0;
  const bySubject={};
  for(const a of list){
    for(const sub of(a.subjects||[])){
      possible++;
      const sc=gradeToScore(a.id,sub,sid);
      if(sc!=null){filled++;sum+=sc;cnt++;bySubject[sub]=bySubject[sub]||[];bySubject[sub].push(sc);}
    }
  }
  const avg=cnt?sum/cnt:0;
  let total=0,present=0,absent=0,late=0,excused=0;
  for(const d in state.attendance||{}){const st=state.attendance[d][sid];if(!st)continue;total++;if(st==='present')present++;else if(st==='absent')absent++;else if(st==='late')late++;else if(st==='excused')excused++;}
  const attRate=total?Math.round(present/total*100):0;
  const recs=(state.discipline||[]).filter(r=>r.sid===sid);
  const pts=recs.reduce((a,b)=>a+b.points,0);
  return{avg,cnt,filled,possible,bySubject,total,present,absent,late,excused,attRate,pts,recs};
}
function classOverallAverage(assessList){
  if(!assessList && state?.aggregates?.classOverallAvg!==undefined) return state.aggregates.classOverallAvg;
  if(!state?.students) return 0;
  const list=assessList||state.assessments;
  let sum=0,cnt=0;
  state.students.forEach(s=>{const st=studentStats(s.id,list);if(st.cnt){sum+=st.avg;cnt++;}});
  return cnt?sum/cnt:0;
}
function subjectAverage(sub,assessList){
  if(!assessList && state?.aggregates?.subjectAvgs?.[sub]!==undefined) return state.aggregates.subjectAvgs[sub];
  if(!state?.students) return 0;
  const list=assessList||state.assessments;
  let sum=0,cnt=0;
  for(const a of list){
    if(!(a.subjects||[]).includes(sub))continue;
    for(const s of state.students){const sc=gradeToScore(a.id,sub,s.id);if(sc!=null){sum+=sc;cnt++;}}
  }
  return cnt?sum/cnt:0;
}
function studentSubjectAvg(sid,sub,assessList){
  const list=assessList||state.assessments;
  let sum=0,cnt=0;
  for(const a of list){
    if(!(a.subjects||[]).includes(sub))continue;
    const sc=gradeToScore(a.id,sub,sid);if(sc!=null){sum+=sc;cnt++;}
  }
  return cnt?sum/cnt:0;
}
function getStudentProgress(sid,assessList){
  const list=assessList||state.assessments;
  const N=state.levels.length;
  const sorted=[...list].sort((a,b)=>(a.date||'').localeCompare(b.date||''));
  const points=[];const classAvgPoints=[];
  sorted.forEach(a=>{
    const subs=a.subjects||[];if(!subs.length)return;
    let sumS=0,cntS=0,sumAll=0,cntAll=0;
    subs.forEach(sub=>{
      const sc=gradeToScore(a.id,sub,sid);
      if(sc!=null){sumS+=sc;cntS++;}
      if(state?.students){
        state.students.forEach(st=>{const s2=gradeToScore(a.id,sub,st.id);if(s2!=null){sumAll+=s2;cntAll++;}});
      }
    });
    if(cntS){points.push({aid:a.id,name:a.name,date:a.date,value:sumS/cntS,max:N});if(cntAll)classAvgPoints.push({aid:a.id,value:sumAll/cntAll});}
  });
  return{points,classAvgPoints,N};
}
function analyzeStudent(sid,assessList){
  const list=assessList||state.assessments;
  const st=studentStats(sid,list);
  const classOverall=classOverallAverage(list);
  const N=state.levels.length;
  const strengths=[],weaknesses=[];
  state.subjects.forEach(sub=>{
    const myAvg=studentSubjectAvg(sid,sub,list);
    if(!myAvg)return;
    const clsAvg=subjectAverage(sub,list);
    const diff=myAvg-clsAvg;
    if(diff>=0.5)strengths.push({sub,myAvg,clsAvg,diff});
    else if(diff<=-0.5)weaknesses.push({sub,myAvg,clsAvg,diff});
  });
  strengths.sort((a,b)=>b.diff-a.diff);
  weaknesses.sort((a,b)=>a.diff-b.diff);
  const{points}=getStudentProgress(sid,list);
  let trend=null,trendDiff=0;
  if(points.length>=3){
    const half=Math.floor(points.length/2);
    const fAvg=points.slice(0,half).reduce((a,b)=>a+b.value,0)/half;
    const sAvg=points.slice(-half).reduce((a,b)=>a+b.value,0)/half;
    trendDiff=sAvg-fAvg;
    if(trendDiff>=0.4)trend='up';else if(trendDiff<=-0.4)trend='down';else trend='stable';
  }
  const attLow=st.total>=5&&st.attRate<80;
  const discLow=st.pts<-2;
  const diffFromClass=st.cnt?st.avg-classOverall:0;
  const recom=[];
  if(trend==='up')recom.push('روند پیشرفت تحصیلی شما صعودی است؛ این مسیر را ادامه دهید. 🌟');
  else if(trend==='down')recom.push('در ارزشیابی‌های اخیر روند نزولی دیده می‌شود؛ با تمرین بیشتر و رفع اشکال، روند را معکوس کنید.');
  else if(trend==='stable')recom.push('عملکرد شما پایدار است؛ برای رشد بیشتر، هدف‌گذاری بلندپروازانه‌تری داشته باشید.');
  if(weaknesses.length)recom.push(`دروس «${weaknesses.slice(0,2).map(w=>w.sub).join('، ')}» نیاز به تمرین و مرور بیشتری دارند.`);
  if(strengths.length)recom.push(`نقاط قوت شما در «${strengths.slice(0,2).map(w=>w.sub).join('، ')}» است؛ از این توانایی برای کمک به دوستان نیز بهره ببرید.`);
  if(attLow)recom.push(`حضور منظم در کلاس تأثیر مستقیمی بر یادگیری دارد. نرخ حضور شما ${fa(st.attRate)}٪ است.`);
  if(discLow)recom.push('توجه به نکات انضباطی و مشارکت سازنده در کلاس، فضا را برای یادگیری بهتر می‌کند.');
  if(st.cnt===0)recom.push('هنوز نمره‌ای برای شما ثبت نشده؛ با معلم خود درباره ارزشیابی‌های پیش‌رو صحبت کنید.');
  if(!recom.length)recom.push('عملکرد متوازن و خوبی دارید. برای حفظ آن، مطالعه منظم روزانه را ادامه دهید.');
  return{st,classOverall,strengths,weaknesses,trend,trendDiff,diffFromClass,attLow,discLow,recom,N};
}

    sub.answers[qid]=value;
  },
  async submitQuiz(auto=false){
    const q=state.quizzes.find(x=>x.id===ui.currentQuizId);
    if(!q)return;
    const sub=q.submissions?.[session.sid];
    if(!sub||sub.submittedAt)return;
    const res=await studentAction('submit_quiz',{quizId:q.id,answers:sub.answers||{}});
    if(!res)return;
    if(auto)toast('⏰ زمان تمام — آزمون به‌صورت خودکار ارسال شد','bad');
    else toast('آزمون ارسال شد','ok');
    ui.tab='s-quizzes';ui.currentQuizId=null;
    renderTabs();renderView();
  }
};

function startQuizTimer(){
  clearInterval(quizTimerInterval);
  quizTimerInterval=setInterval(()=>{
    if(ui.tab!=='s-quiz-taking'){clearInterval(quizTimerInterval);return;}
    const q=state.quizzes.find(x=>x.id===ui.currentQuizId);
    if(!q){clearInterval(quizTimerInterval);return;}
    const sub=q.submissions?.[session.sid];
    if(!sub||sub.submittedAt){clearInterval(quizTimerInterval);return;}
    if(!q.durationMin)return;
    const elapsed=(Date.now()-new Date(sub.startedAt).getTime())/1000;
    const remaining=q.durationMin*60-elapsed;
    const el=document.getElementById('quizTimer');
    if(!el)return;
    if(remaining<=0){el.textContent='⏰ زمان تمام';el.className='quiz-timer-badge danger';App.submitQuiz(true);clearInterval(quizTimerInterval);return;}
    const m=Math.floor(remaining/60);const s=Math.floor(remaining%60);
    el.textContent=`⏱️ ${fa(String(m).padStart(2,'0'))}:${fa(String(s).padStart(2,'0'))}`;
    el.className='quiz-timer-badge '+(remaining<60?'danger':remaining<300?'warn':'');
  },1000);
}

/* ==================== CHART ==================== */
function renderProgressChart(sid,assessList,{showClass=true,height=220}={}){
  const{points,classAvgPoints,N}=getStudentProgress(sid,assessList);
  if(!points.length)return `<div class="chart-empty">نموداری برای نمایش نیست</div>`;
  const W=600,H=height;const padL=52,padR=20,padT=26,padB=44;
  const innerW=W-padL-padR,innerH=H-padT-padB;
  const n=points.length;
  const xFor=i=>n===1?padL+innerW/2:padL+(i/(n-1))*innerW;
  const yFor=v=>padT+innerH-(v/N)*innerH;
  const lineD=points.map((p,i)=>`${i?'L':'M'}${xFor(i).toFixed(1)},${yFor(p.value).toFixed(1)}`).join(' ');
  const areaD=lineD+` L${xFor(n-1).toFixed(1)},${(padT+innerH).toFixed(1)} L${xFor(0).toFixed(1)},${(padT+innerH).toFixed(1)} Z`;
  let classLine='';
  if(showClass&&classAvgPoints.length===n&&n>1){classLine=classAvgPoints.map((p,i)=>`${i?'L':'M'}${xFor(i).toFixed(1)},${yFor(p.value).toFixed(1)}`).join(' ');}
  const gridLines=[];
  for(let k=0;k<=N;k++){
    const y=yFor(k);
    gridLines.push(`<line x1="${padL}" y1="${y.toFixed(1)}" x2="${W-padR}" y2="${y.toFixed(1)}" stroke="#e5e7eb" stroke-width="1" stroke-dasharray="4,4"/>`);
    const level=state.levels[N-k];
    if(level)gridLines.push(`<text x="${padL-6}" y="${(y+4).toFixed(1)}" font-size="10" font-weight="700" fill="${level.color}" text-anchor="end">${esc(level.label)}</text>`);
  }
  const dots=points.map((p,i)=>{
    const lv=levelFromScore(p.value);
    const color=lv?lv.color:'#7c3aed';
    return `<circle cx="${xFor(i).toFixed(1)}" cy="${yFor(p.value).toFixed(1)}" r="5.5" fill="#fff" stroke="${color}" stroke-width="3"/>`;
  }).join('');
  const valueLabels=points.map((p,i)=>`<text x="${xFor(i).toFixed(1)}" y="${(yFor(p.value)-11).toFixed(1)}" font-size="10.5" font-weight="800" fill="#4c1d95" text-anchor="middle">${fa(p.value.toFixed(1))}</text>`).join('');
  const xLabels=points.map((p,i)=>`<text x="${xFor(i).toFixed(1)}" y="${H-14}" font-size="9.5" font-weight="700" fill="#64748b" text-anchor="middle">${p.date?isoToShortJalali(p.date):''}</text>`).join('');
  return `<svg viewBox="0 0 ${W} ${H}" class="chart-svg">
    <defs><linearGradient id="areaGrad-${sid}" x1="0" x2="0" y1="0" y2="1">
      <stop offset="0%" stop-color="#7c3aed" stop-opacity=".35"/>
      <stop offset="100%" stop-color="#7c3aed" stop-opacity="0"/>
    </linearGradient></defs>
    ${gridLines.join('')}
    <path d="${areaD}" fill="url(#areaGrad-${sid})"/>
    ${classLine?`<path d="${classLine}" fill="none" stroke="#94a3b8" stroke-width="2" stroke-dasharray="6,5" opacity=".8"/>`:''}
    <path d="${lineD}" fill="none" stroke="#7c3aed" stroke-width="3" stroke-linejoin="round" stroke-linecap="round"/>
    ${dots}${valueLabels}${xLabels}
  </svg>`;
}

function analysisBoxHTML(sid,assessList){
  const an=analyzeStudent(sid,assessList);
  const{st,trend,strengths,weaknesses,attLow,discLow,recom,diffFromClass,N}=an;
  let mood='info';
  if(st.cnt===0)mood='info';
  else if(st.avg>=N-1&&!attLow&&!discLow)mood='good';
  else if(st.avg<2||attLow||discLow)mood='bad';
  else mood='warn';
  const trendMap={
    up:{icon:'📈',cls:'g',text:'روند رو به رشد'},
    down:{icon:'📉',cls:'r',text:'روند نزولی'},
    stable:{icon:'➡️',cls:'y',text:'روند پایدار'},
    null:{icon:'—',cls:'',text:'داده کافی نیست'}
  };
  const t=trendMap[trend]||trendMap[null];
  const diffCls=diffFromClass>0.3?'g':diffFromClass<-0.3?'r':'y';
  const diffText=diffFromClass>0.3?'بالاتر از میانگین':diffFromClass<-0.3?'پایین‌تر از میانگین':'در سطح میانگین';
  return `<div class="analysis-box ${mood}">
    <div class="an-title">🔍 تحلیل وضعیت تحصیلی</div>
    <div style="display:flex;gap:6px;flex-wrap:wrap;margin-bottom:10px">
      <span class="chip ${t.cls}">${t.icon} ${t.text}</span>
      <span class="chip ${diffCls}">${diffText}</span>
      <span class="chip ${st.attRate>=90?'g':st.attRate>=70?'y':'r'}">✅ حضور ${fa(st.attRate)}٪</span>
      <span class="chip ${st.pts>=0?'g':'r'}">⭐ انضباط ${fa(st.pts)}</span>
    </div>
    ${strengths.length?`<ul>${strengths.slice(0,3).map(s=>`<li><b>نقطه قوت:</b> ${esc(s.sub)} — شما ${fa(s.myAvg.toFixed(2))} در برابر کلاس ${fa(s.clsAvg.toFixed(2))}</li>`).join('')}</ul>`:''}

let lockTimeout;
function lockScreen(){if(!state?.antiScreenshot)return;if(!session)return;$('#lockOverlay').classList.remove('hidden');}
function unlockScreen(){$('#lockOverlay').classList.add('hidden');}
function setupAntiScreenshot(){
  if(!state?.antiScreenshot)return;
  document.addEventListener('contextmenu',e=>{e.preventDefault();});
  document.addEventListener('keydown',e=>{
    const k=e.key.toLowerCase();
    if((e.ctrlKey||e.metaKey)&&['p','s','u'].includes(k)){e.preventDefault();toast('⛔ مجاز نیست','bad');return;}
    if(e.key==='F12'){e.preventDefault();toast('⛔ Developer Tools غیرفعال','bad');return;}
    if(e.key==='PrintScreen'||e.code==='PrintScreen'){
      e.preventDefault();
      try{navigator.clipboard.writeText('');}catch(err){}
      lockScreen();toast('⛔ اسکرین‌شات غیرفعال','bad');
      setTimeout(unlockScreen,1200);
    }
  });
  document.addEventListener('visibilitychange',()=>{
    if(document.hidden)lockScreen();else setTimeout(unlockScreen,200);
  });
  window.addEventListener('blur',()=>{clearTimeout(lockTimeout);lockTimeout=setTimeout(lockScreen,400);});
  window.addEventListener('focus',()=>{clearTimeout(lockTimeout);setTimeout(unlockScreen,200);});
  setInterval(()=>{if(!document.hasFocus()&&session)lockScreen();},3000);
}
